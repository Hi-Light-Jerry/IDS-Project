Classify working and non-working days

# Compare hours when the bike service was operating.
bikes_open <- bikes %>%
  filter(functioning == "Yes") %>%
  mutate(
    weekday = wday(date),
    # Public holidays belong to the non-working group even when they fall on a weekday.
    day_type = if_else(
      weekday == 1 | weekday == 7 | holiday == "Holiday",
      "Non-working day",
      "Working day"
    )
  )

count(bikes_open, day_type)

Calculate mean hourly rentals

# Keep the sample size because the two day types contain different numbers of observations.
hourly_rentals <- bikes_open %>%
  group_by(season, day_type, hour) %>%
  summarise(
    n_hours = n(),
    mean_rentals = mean(rentals)
  )

Compare hourly rental patterns across seasons

# Use the same vertical scale in all four panels so rental levels can be compared across seasons.
ggplot(
  data = hourly_rentals,
  mapping = aes(x = hour, y = mean_rentals, colour = day_type)
) +
  geom_line() +
  facet_wrap(~ season, ncol = 2) +
  labs(
    x = "Hour of day",
    y = "Mean rentals per hour",
    colour = "Day type"
  )
